打开VB，新建ActiveX Dll,将工程改名为COMConnection,类模块改名为class,在class的代码里写

Option Explicit

  
Public Function GetConn(m\_objConn As Connection) As String  
      
    On Error GoTo here  
      
    Dim m\_szConnstr As String  
      
    m\_szConnstr = "Provider=SQLOLEDB.1;Data Source=192.168.0.101;User ID=sa;Password=;Initial Catalog=aaa"  
      
    m\_objConn.Open m\_szConnstr  
      
    GetConn = "OK"  
    Exit Function  
      
here:  
    GetConn = "open database is error.error id is " & err.Description

End Function

  
然后编译  
Asp里这样用

    dim obj,conn,msg  
    set obj=server.create("COMConnection.class")  
    set conn=server.create("adodb.connection")

        //这里写代码，conn已经打开数据库了

    end if  
%>
