Function OCXTest()  
OCXTest = 1   
dim ocx  
on error resume next  
err.clear  
set ocx = createobject("I\_FLEXGRID.FlexGridCtrl.2")

if err <> 0 then  
OCXTest = 0  
err.clear  
exit function  
end if   
set ocx = createobject("LEDGER50\_I.Ledger50Ctrl\_I.1")

OCXTest = 0  
err.clear  
exit function  
end if   
  
End function  
  
//-->  
</SCRIPT>
