{{- $prefix := "" -}} {{- if ne .key "main" -}} {{- $prefix = printf "/%s" .key -}} {{- end -}} {{- $segment := cond (eq .taxonomy "categories") "cat" "tag" -}} {{- $stem := printf "%s/%s/%s" $prefix $segment .slug -}} {{- $page := .page -}} {{- $total := .total -}} {{- $window := site.Params.pagerWindow | default 4 -}} {{- if gt $total 1 -}} {{- $start := sub $page $window -}} {{- if lt $start 1 }}{{ $start = 1 }}{{ end -}} {{- $end := add $page $window -}} {{- if gt $end $total }}{{ $end = $total }}{{ end -}}
